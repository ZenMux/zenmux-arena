% ---- A boxed "key finding" callout, reused throughout the results. ---------
% Needs only xcolor (no tcolorbox/framed/mdframed, which BasicTeX lacks). A tinted
% panel with a thick left accent rule and an inline bold title. Built from a
% minipage in a \colorbox, so it does not break across a page — findings are kept
% short by design. The \savebox lets the panel width track the text column.
\newsavebox{\findingbox}
\newenvironment{finding}[1][发现]%
  {\par\smallskip\noindent
   \begin{lrbox}{\findingbox}%
   \begin{minipage}{\dimexpr\linewidth-2.4em\relax}%
   \smallskip{\bfseries\color{accent}#1.\ }\ignorespaces}%
  {\smallskip\end{minipage}\end{lrbox}%
   {\color{accent!8}\rule[-0.4ex]{0.35em}{\dimexpr\ht\findingbox+\dp\findingbox\relax}}%
   \hspace{-0.35em}%
   {\color{accent}\rule[-0.4ex]{0.18em}{\dimexpr\ht\findingbox+\dp\findingbox\relax}}%
   \colorbox{accent!5}{\usebox{\findingbox}}\par\smallskip}

% ---- Hyperlinks (loaded late). ---------------------------------------------

% ---- Inline helper macros. -------------------------------------------------
% A compact percentage that keeps prose readable: \pc{12.9} -> 12.9%
\newcommand{\pc}[1]{#1\,\%}
% Shorthand for a model/vendor name set in upright text inside Chinese prose.
\newcommand{\name}[1]{\textsf{#1}}

% A long model id in typewriter that may wrap: \texttt with \allowbreak after the
% structural separators. Kept deliberately simple — the caller marks break points,
% e.g. \modelid{anthropic}{/claude-opus-4.8}{:anthropic}. Each braced group is glued
% with an \allowbreak between, so the id can wrap at the slashes/colons but never
% mid-token. (Named \modelid, not \mid — \mid is a built-in math relation.)
\newcommand{\modelid}[3]{\texttt{#1\allowbreak#2\allowbreak#3}}
